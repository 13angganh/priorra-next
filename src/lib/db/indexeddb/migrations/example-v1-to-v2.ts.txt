import type { IDBPTransaction } from "idb";
import type { PriorraDBSchema } from "../client";
import { STORE_NAMES } from "../schema";

export async function migrateV1ToV2(
  tx: IDBPTransaction<PriorraDBSchema, string[], "versionchange">
): Promise<void> {
  const taskStore = tx.objectStore(STORE_NAMES.tasks);
  let cursor = await taskStore.openCursor();

  while (cursor) {
    const task = cursor.value;
    // Hypothetical: field didn't exist in v1, so it's missing here,
    // not just empty — this is what a real "the shape actually
    // changed" migration looks like, as opposed to a value change.
    if (!("tags" in task)) {
      await cursor.update({ ...task, tags: [] });
    }
    cursor = await cursor.continue();
  }
}
